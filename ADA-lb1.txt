/* 1. TOPOLOGICAL SORT */

#include<stdio.h>

int main()
{
    int n,a[20][20],indeg[20]={0},i,j,k,count=0;

    printf("Enter number of vertices: ");
    scanf("%d",&n);

    printf("Enter adjacency matrix:\n");
    for(i=0;i<n;i++)
        for(j=0;j<n;j++)
            scanf("%d",&a[i][j]);

    for(i=0;i<n;i++)
        for(j=0;j<n;j++)
            if(a[i][j])
                indeg[j]++;

    printf("Topological Order: ");

    while(count<n)
    {
        for(i=0;i<n;i++)
        {
            if(indeg[i]==0)
            {
                printf("%d ",i);
                indeg[i]=-1;
                count++;

                for(k=0;k<n;k++)
                    if(a[i][k])
                        indeg[k]--;
            }
        }
    }
    return 0;
}

/* 2. JOHNSON TROTTER ALGORITHM */

#include<stdio.h>

int mobile(int a[],int dir[],int n)
{
    int mobile_prev=0,mobile_index=-1,i;

    for(i=0;i<n;i++)
    {
        if(dir[a[i]-1]==0 && i!=0 && a[i]>a[i-1] && a[i]>mobile_prev)
        {
            mobile_prev=a[i];
            mobile_index=i;
        }

        if(dir[a[i]-1]==1 && i!=n-1 && a[i]>a[i+1] && a[i]>mobile_prev)
        {
            mobile_prev=a[i];
            mobile_index=i;
        }
    }
    return mobile_index;
}

int main()
{
    int n,i,j,pos,temp;
    printf("Enter n: ");
    scanf("%d",&n);

    int a[n],dir[n];

    for(i=0;i<n;i++)
    {
        a[i]=i+1;
        dir[i]=0;
    }

    for(i=0;i<n;i++)
        printf("%d ",a[i]);
    printf("\n");

    while(1)
    {
        pos=mobile(a,dir,n);

        if(pos==-1)
            break;

        if(dir[a[pos]-1]==0)
        {
            temp=a[pos];
            a[pos]=a[pos-1];
            a[pos-1]=temp;
            pos--;
        }
        else
        {
            temp=a[pos];
            a[pos]=a[pos+1];
            a[pos+1]=temp;
            pos++;
        }

        for(j=0;j<n;j++)
            if(a[j]>a[pos])
                dir[a[j]-1]=1-dir[a[j]-1];

        for(j=0;j<n;j++)
            printf("%d ",a[j]);
        printf("\n");
    }
    return 0;
}

/* 3. MERGE SORT */

#include<stdio.h>
#include<time.h>

void merge(int a[],int l,int m,int r)
{
    int i=l,j=m+1,k=0,temp[1000];

    while(i<=m && j<=r)
    {
        if(a[i]<a[j]) temp[k++]=a[i++];
        else temp[k++]=a[j++];
    }

    while(i<=m) temp[k++]=a[i++];
    while(j<=r) temp[k++]=a[j++];

    for(i=l,k=0;i<=r;i++,k++)
        a[i]=temp[k];
}

void mergesort(int a[],int l,int r)
{
    if(l<r)
    {
        int m=(l+r)/2;
        mergesort(a,l,m);
        mergesort(a,m+1,r);
        merge(a,l,m,r);
    }
}

int main()
{
    int n,i,a[1000];
    clock_t start,end;

    scanf("%d",&n);

    for(i=0;i<n;i++)
        scanf("%d",&a[i]);

    start=clock();
    mergesort(a,0,n-1);
    end=clock();

    printf("Sorted Array:\n");

    for(i=0;i<n;i++)
        printf("%d ",a[i]);

    printf("\nTime=%lf",(double)(end-start)/CLOCKS_PER_SEC);

    return 0;
}

/* 4. QUICK SORT */

#include<stdio.h>
#include<time.h>

int partition(int a[],int low,int high)
{
    int pivot=a[low],i=low+1,j=high,temp;

    while(i<=j)
    {
        while(i<=high && a[i]<=pivot) i++;
        while(a[j]>pivot) j--;

        if(i<j)
        {
            temp=a[i];
            a[i]=a[j];
            a[j]=temp;
        }
    }

    temp=a[low];
    a[low]=a[j];
    a[j]=temp;

    return j;
}

void quicksort(int a[],int low,int high)
{
    if(low<high)
    {
        int p=partition(a,low,high);
        quicksort(a,low,p-1);
        quicksort(a,p+1,high);
    }
}

int main()
{
    int n,i,a[1000];
    clock_t start,end;

    scanf("%d",&n);

    for(i=0;i<n;i++)
        scanf("%d",&a[i]);

    start=clock();
    quicksort(a,0,n-1);
    end=clock();

    for(i=0;i<n;i++)
        printf("%d ",a[i]);

    printf("\nTime=%lf",(double)(end-start)/CLOCKS_PER_SEC);

    return 0;
}

/* 5. HEAP SORT */

#include<stdio.h>
#include<time.h>

void heapify(int a[],int n,int i)
{
    int largest=i,l=2*i+1,r=2*i+2,temp;

    if(l<n && a[l]>a[largest]) largest=l;
    if(r<n && a[r]>a[largest]) largest=r;

    if(largest!=i)
    {
        temp=a[i];
        a[i]=a[largest];
        a[largest]=temp;

        heapify(a,n,largest);
    }
}

int main()
{
    int n,a[1000],i,temp;
    clock_t start,end;

    scanf("%d",&n);

    for(i=0;i<n;i++)
        scanf("%d",&a[i]);

    start=clock();

    for(i=n/2-1;i>=0;i--)
        heapify(a,n,i);

    for(i=n-1;i>0;i--)
    {
        temp=a[0];
        a[0]=a[i];
        a[i]=temp;

        heapify(a,i,0);
    }

    end=clock();

    for(i=0;i<n;i++)
        printf("%d ",a[i]);

    printf("\nTime=%lf",(double)(end-start)/CLOCKS_PER_SEC);

    return 0;
}

/* 6. 0/1 KNAPSACK */

#include<stdio.h>

int max(int a,int b)
{
    return a>b?a:b;
}

int main()
{
    int n,w[20],p[20],W,i,j,k[20][20];

    scanf("%d",&n);

    for(i=1;i<=n;i++)
        scanf("%d",&w[i]);

    for(i=1;i<=n;i++)
        scanf("%d",&p[i]);

    scanf("%d",&W);

    for(i=0;i<=n;i++)
    {
        for(j=0;j<=W;j++)
        {
            if(i==0 || j==0)
                k[i][j]=0;
            else if(w[i]<=j)
                k[i][j]=max(k[i-1][j],p[i]+k[i-1][j-w[i]]);
            else
                k[i][j]=k[i-1][j];
        }
    }

    printf("Maximum Profit=%d",k[n][W]);

    return 0;
}

/* 9. KRUSKAL'S ALGORITHM */

#include<stdio.h>

int parent[20];

int find(int i)
{
    while(parent[i])
        i=parent[i];
    return i;
}

int uni(int i,int j)
{
    if(i!=j)
    {
        parent[j]=i;
        return 1;
    }
    return 0;
}

int main()
{
    int n,a[20][20],i,j,min,cost=0,u,v,e=1;

    scanf("%d",&n);

    for(i=0;i<n;i++)
        for(j=0;j<n;j++)
            scanf("%d",&a[i][j]);

    while(e<n)
    {
        min=999;

        for(i=0;i<n;i++)
            for(j=0;j<n;j++)
                if(a[i][j]<min)
                {
                    min=a[i][j];
                    u=i;
                    v=j;
                }

        i=find(u);
        j=find(v);

        if(uni(i,j))
        {
            printf("%d-%d=%d\n",u,v,min);
            cost+=min;
            e++;
        }

        a[u][v]=a[v][u]=999;
    }

    printf("Cost=%d",cost);

    return 0;
}

/* 10. FRACTIONAL KNAPSACK */

#include<stdio.h>

int main()
{
    int n,i,j;
    float w[20],p[20],ratio[20],cap,temp,maxprofit=0;

    scanf("%d",&n);

    for(i=0;i<n;i++)
        scanf("%f %f",&p[i],&w[i]);

    scanf("%f",&cap);

    for(i=0;i<n;i++)
        ratio[i]=p[i]/w[i];

    for(i=0;i<n-1;i++)
    {
        for(j=i+1;j<n;j++)
        {
            if(ratio[i]<ratio[j])
            {
                temp=ratio[i]; ratio[i]=ratio[j]; ratio[j]=temp;
                temp=p[i]; p[i]=p[j]; p[j]=temp;
                temp=w[i]; w[i]=w[j]; w[j]=temp;
            }
        }
    }

    for(i=0;i<n;i++)
    {
        if(cap>=w[i])
        {
            maxprofit+=p[i];
            cap-=w[i];
        }
        else
        {
            maxprofit+=ratio[i]*cap;
            break;
        }
    }

    printf("Maximum Profit=%f",maxprofit);

    return 0;
}

/* 11. DIJKSTRA'S ALGORITHM */

#include<stdio.h>

int main()
{
    int n,cost[20][20],dist[20],visited[20];
    int i,j,u,min,start;

    scanf("%d",&n);

    for(i=0;i<n;i++)
        for(j=0;j<n;j++)
            scanf("%d",&cost[i][j]);

    scanf("%d",&start);

    for(i=0;i<n;i++)
    {
        dist[i]=cost[start][i];
        visited[i]=0;
    }

    visited[start]=1;

    for(i=1;i<n;i++)
    {
        min=999;

        for(j=0;j<n;j++)
        {
            if(!visited[j] && dist[j]<min)
            {
                min=dist[j];
                u=j;
            }
        }

        visited[u]=1;

        for(j=0;j<n;j++)
        {
            if(!visited[j] && dist[u]+cost[u][j]<dist[j])
                dist[j]=dist[u]+cost[u][j];
        }
    }

    for(i=0;i<n;i++)
        printf("%d -> %d = %d\n",start,i,dist[i]);

    return 0;
}

/* 12. N QUEENS */

#include<stdio.h>
#include<stdlib.h>

int x[20],count=0;

int place(int k,int i)
{
    int j;

    for(j=1;j<k;j++)
        if(x[j]==i || abs(x[j]-i)==abs(j-k))
            return 0;

    return 1;
}

void nqueen(int k,int n)
{
    int i,j;

    for(i=1;i<=n;i++)
    {
        if(place(k,i))
        {
            x[k]=i;

            if(k==n)
            {
                count++;

                printf("\nSolution %d:\n",count);

                for(j=1;j<=n;j++)
                    printf("%d ",x[j]);
            }
            else
                nqueen(k+1,n);
        }
    }
}

int main()
{
    int n;

    scanf("%d",&n);

    nqueen(1,n);

    return 0;
}

# Short Viva Derivations

## 1. Fractional Knapsack

Main work is sorting items by profit/weight ratio.

[
O(n \log n)
]

Selecting items takes:

[
O(n)
]

Total:

[
O(n \log n)
]

---

## 2. Topological Sort

Using adjacency matrix:

* Compute indegrees → O(V²)
* Process vertices → O(V²)

Total:

[
O(V^2)
]

---

## 3. Merge Sort

Recurrence:

[
T(n)=2T(n/2)+n
]

Master Theorem:

[
T(n)=O(n\log n)
]

---

## 4. Quick Sort

Average:

[
T(n)=2T(n/2)+n
]

[
O(n\log n)
]

Worst case:

[
T(n)=T(n-1)+n
]

[
O(n^2)
]

---

## 5. 0/1 Knapsack

DP table size:

[
(n+1)(W+1)
]

Each cell computed once.

[
O(nW)
]

For the given example:

* n = 3
* W = 50

Operations ≈

[
3 \times 50 = 150
]

---

## 6. Kruskal's Algorithm

Sorting edges dominates.

[
O(E \log E)
]

Union-Find cost is comparatively smaller.

---

## 7. Floyd's Algorithm

Three nested loops:

```c
for(k)
 for(i)
  for(j)
```

[
O(V^3)
]

---

## 8. Prim's Algorithm

Using adjacency matrix:

[
O(V^2)
]

(Using Min Heap → O(E logV), but lab programs use matrix.)

---

## 9. N-Queens

Choices reduce row by row:

[
N\times(N-1)\times(N-2)\cdots
]

[
O(N!)
]

---

## 10. Heap Sort

Build Heap:

[
O(n)
]

Heapify repeatedly:

[
n\times O(\log n)
]

Total:

[
O(n\log n)
]

Here are the standard algorithms and algorithmic design paradigms used to solve each of the problems listed on your Analysis and Design of Algorithms (ADA) lab exam sheet.
## 1. GPS Navigation System
> **Problem:** Find the shortest path from a source location to all other locations.
> 
 * **Paradigm:** Greedy Method
 * **Algorithm:** **Dijkstra’s Algorithm**
 * **Explanation:** Since it requires finding the shortest path from a single source to all other nodes (locations) in a network with non-negative edge weights, Dijkstra's algorithm is the ideal choice. If the graph contains negative weights, **Bellman-Ford Algorithm** (Dynamic Programming) would be used instead.
## 2. Library Book Catalog Organization
> **Problem:** Arrange accession numbers in ascending order.
> 
 * **Paradigm:** Divide and Conquer (or Comparison-Based Sorting)
 * **Algorithm:** **Merge Sort** or **Quick Sort**
 * **Explanation:** Organizing a catalog requires an efficient sorting mechanism. Merge Sort offers a stable O(n \log n) time complexity, making it excellent for large data sets, while Quick Sort is highly efficient in practice with an average time complexity of O(n \log n).
## 3. Cargo Loading Optimization
> **Problem:** Maximize profit by selecting full or partial packages within a fixed capacity.
> 
 * **Paradigm:** Greedy Method
 * **Algorithm:** **Fractional Knapsack Algorithm**
 * **Explanation:** The key phrase here is **"partial packages"**. Because you are allowed to take fractions of an item to maximize profit within a capacity limit, a greedy approach (sorting items by their profit-to-weight ratio) yields the absolute optimal solution.
## 4. Student Result Processing
> **Problem:** Sort student marks efficiently and record the execution time for different input sizes.
> 
 * **Paradigm:** Divide and Conquer / Decrease and Conquer
 * **Algorithm:** **Merge Sort**, **Quick Sort**, or **Heap Sort**
 * **Explanation:** This is a classic lab experiment designed to analyze empirical time complexity. You typically implement a sorting algorithm (like Quick Sort or Merge Sort), generate random datasets of varying sizes (N = 1000, 5000, 10000), use system time functions to measure the execution time, and plot a graph to verify the theoretical O(n \log n) complexity.
## 5. Optical Fiber Network Design
> **Problem:** Connect all offices with minimum cable installation cost.
> 
 * **Paradigm:** Greedy Method
 * **Algorithm:** **Prims’s Algorithm** or **Kruskal’s Algorithm**
 * **Explanation:** Connecting all nodes in a network with the minimum possible total edge weight without creating any cycles is the definition of finding a **Minimum Spanning Tree (MST)**. Both Prim's and Kruskal's algorithms solve this efficiently.
## 6. Course Scheduling System
> **Problem:** Given courses and prerequisite relationships, determine a valid order in which all courses can be completed.
> 
 * **Paradigm:** Decrease and Conquer / Graph Traversal
 * **Algorithm:** **Topological Sorting** (using Kahn's Algorithm or Depth-First Search)
 * **Explanation:** Prerequisite constraints form a Directed Acyclic Graph (DAG). Topological sorting provides a linear ordering of vertices such that for every directed edge u \to v, node u comes before v in the ordering.
## 7. Hospital Patient Priority Management
> **Problem:** Sort patient severity scores and record execution time.
> 
 * **Paradigm:** Decrease and Conquer / Heap Structures
 * **Algorithm:** **Heap Sort** (or utilizing a **Priority Queue**)
 * **Explanation:** In hospital priority management, patients with higher severity scores must be treated first. A Max-Heap or Heap Sort is perfectly suited for managing priority elements dynamically. Like question 4, this involves tracking execution times across different inputs.
## 8. Security Camera Placement
> **Problem:** Place N cameras on an N \times N grid so that no two interfere.
> 
 * **Paradigm:** Backtracking
 * **Algorithm:** **N-Queens Problem Algorithm**
 * **Explanation:** This is a direct real-world adaptation of the classic N-Queens problem. You attempt to place a camera in a row, check for conflicts (horizontal, vertical, and diagonal interference), and if a conflict arises, you backtrack to the previous step and try a different position.
## 9. Airline Route Planner
> **Problem:** Determine the minimum travel cost between every pair of cities.
> 
 * **Paradigm:** Dynamic Programming
 * **Algorithm:** **Floyd-Warshall Algorithm**
 * **Explanation:** The problem asks for the minimum cost between **every pair** of cities, which means it requires an **All-Pairs Shortest Path** solution. The Floyd-Warshall algorithm computes this efficiently for a graph using a dynamic programming table in O(V^3) time complexity.
## 10. Seating Arrangement Generator
> **Problem:** Generate all possible seating arrangements of N guests exactly once.
> 
 * **Paradigm:** Brute Force / Backtracking
